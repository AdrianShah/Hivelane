import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

```bash
pipx install git+https://github.com/AdrianShah/Hivelane   # not on PyPI yet
hivelane up                # in your repo: hooks, room, background daemon
```

<CardGrid>
  <Card title="Pre-edit checks" icon="approve-check">
    Agent hooks ask the local daemon before every write. The relay answers
    allow, warn or block from what the rest of the room is doing, down to the
    function or line range.
  </Card>
  <Card title="Every agent, one room" icon="puzzle">
    Claude Code, Codex, Cursor and Gemini CLI through native hooks; Aider and
    anything else through `hivelane run`; humans through the VS Code extension.
  </Card>
  <Card title="Shared dashboard" icon="laptop">
    One lane per agent with intent, plans, captions and diffs, plus every warn
    and block decision with its reason.
  </Card>
  <Card title="Self-hostable" icon="seti:docker">
    The relay is one small FastAPI process in a container. Run it on your own
    host and nothing leaves your network.
  </Card>
</CardGrid>
